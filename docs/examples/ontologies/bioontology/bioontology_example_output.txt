"""
Example: Using the BioOntology Adapter for Biomedical Concept Lookup
"""

from knowledge_lookup import LookupConfig, create_knowledge_lookup
from knowledge_lookup.models import KnowledgeSource


async def main():
    # Note: BioOntology adapter may require an API key
    # Set BIOPORTAL_API_KEY environment variable
    # os.environ['BIOPORTAL_API_KEY'] = 'your_api_key_here'
    
    config = LookupConfig(
        max_results_per_source=20,
        timeout_per_source=30.0,
        enabled_sources=[KnowledgeSource.BIOONTOLOGY],
    )
    
    lookup = create_knowledge_lookup(config=config)
    
    print(f"Checking if BioOntology adapter is available...")
    if lookup:
        print(f"  ✓ BioOntology adapter is available")
    else:
        print(f"  ✗ BioOntology adapter is not available")
        return
    
    query = "cancer"

    print(f"\Searching for 'cancer' concepts in BioOntology...")
    results = await lookup.search_concepts(query, sources=[KnowledgeSource.BIOONTOLOGY])
    

    print(f"\Found {len(results)} results from BioOntology:")
    for i, result in enumerate(results[:5]):

        print(f"\  Result {i+1}:")
        print(f"    Concept: {result.primary_label}")
        print(f"    ID: {result.primary_id}")
        print(f"    Source: {result.source}")
        print(f"    Type: {result.concept_type}")
        
        if result.mappings:
            print(f"    Mappings ({len(result.mappings)}):")
            for mapping in result.mappings[:3]:
                print(f"      - {mapping.source_id}: {mapping.source_label}")
        
        if i == 0 and result.primary_id:

            print(f"\  Getting detailed information for: {result.primary_id}")
            try:
                details = await lookup.get_concept_details(result.primary_id, sources=[KnowledgeSource.BIOONTOLOGY])
                if details:
                    print(f"    Details found:")
                    print(f"      Label: {details.primary_label}")
                    print(f"      Type: {details.concept_type}")
                else:
                    print(f"    No detailed information available")
            except Exception as e:
                print(f"    Error getting details: {e}")
    else:
        print("  No results found for 'cancer'")


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
