"""
Example: Using the DisGeNET Adapter for Biomedical Concept Lookup
"""

from knowledge_lookup import LookupConfig, create_knowledge_lookup
from knowledge_lookup.models import KnowledgeSource


async def main():
    # Note: DisGeNET adapter may require an API key
    # Set DISGENET_API_KEY environment variable
    # os.environ['DISGENET_API_KEY'] = 'your_api_key_here'
    
    config = LookupConfig(
        max_results_per_source=20,
        timeout_per_source=30.0,
        enabled_sources=[KnowledgeSource.DISGENET],
    )
    
    lookup = create_knowledge_lookup(config=config)
    
    print(f"Checking if DisGeNET adapter is available...")
    if lookup:
        print(f"  ✓ DisGeNET adapter is available")
    else:
        print(f"  ✗ DisGeNET adapter is not available")
        return
    
    query = "cancer"
    print(f"\\nSearching for '{query}' concepts in DisGeNET...")
    results = await lookup.search_concepts(query, sources=[KnowledgeSource.DISGENET])
    
    print(f"\\nFound {len(results)} results from DisGeNET:")
    for i, result in enumerate(results[:5]):
        print(f"\\n  Result {i+1}:")
        print(f"    Concept: {result.primary_label}")
        print(f"    ID: {result.primary_id}")
        print(f"    Source: {result.source}")
        print(f"    Type: {result.concept_type}")
        
        if result.mappings:
            print(f"    Mappings ({len(result.mappings)}):")
            for mapping in result.mappings[:3]:
                print(f"      - {mapping.source_id}: {mapping.source_label}")
        
        if i == 0 and result.primary_id:
            print(f"\\n  Getting detailed information for: {result.primary_id}")
            try:
                details = await lookup.get_concept_details(result.primary_id, sources=[KnowledgeSource.DISGENET])
                if details:
                    print(f"    Details found:")
                    print(f"      Label: {details.primary_label}")
                    print(f"      Type: {details.concept_type}")
                else:
                    print(f"    No detailed information available")
            except Exception as e:
                print(f"    Error getting details: {e}")
    else:
        print("  No results found for '{query}'")


if __name__ == "__main__":
    import asyncio
    asyncio.run(main())
